# ===========================================================================
#  config.txt —— 程序配置（本文件可以不删，改完重新打开程序即可生效）
#  说明：所有项都可以删；删掉的项会自动用程序内置的默认值。
# ===========================================================================

# 模型名：deepseek-flash = 支持图片（默认，推荐）
#         deepseek-v4-pro = 更聪明但不支持图片
model = deepseek-flash

# 接口地址（一般不用改）
base_url = https://api.deepseek.com

# 单次请求超时（秒）
timeout_sec = 180

# 是否默认开启“深度思考”：off = 先答再说（快）；on = 先想再答（慢但难题准）
thinking = off

# 思考强度（thinking=on 时生效）：low / high / max
reasoning_effort = high

# 历史对话里最多保留几张图的原始数据（省钱用，1 就够了）
keep_image_turns = 1

# 界面模式：
#   compact = 极简两行（默认）：只有一行输出 + 一行输入；挂了图片时输入行右边
#             出现“移除图片”按钮。发送/停止/清空/思考开关/重新读取/编辑提示词
#             全部收进右键菜单，功能一个不少。
#   full    = 完整界面：大对话区 + 一排按钮 + 状态栏 + 缩略图预览。
ui = full

# 窗口默认宽度（逻辑像素，以 96dpi 为基准；屏幕缩放 150% 时实际显示会乘 1.5）
# 极简模式下窗口高度是按两行文字自动算出来的，并且锁死（不允许上下拉伸），
# 所以 window_height 只在 ui = full 时生效。
window_width = 500
window_height = 400

# 送回去的历史消息条数上限（越大越记得住，也越贵）
# 注意：提示词（提示词.txt）每一轮都会单独发一次，不占这个额度，也不会被它挤掉。
#       这个值只管“模型记得住多少轮对话”：
#         3  = 只记得上一轮（很省，但问新题时模型看不到更早的内容，容易像“失忆”）
#         20 = 记得最近 10 轮左右（日常推荐）
#         0  = 不限制（聊很久会越来越贵）
# 成本参考：上限放到 40，每轮多送的历史也就几百到两千字；deepseek-flash 非高峰
#          输入价 $0.15/百万 token，约合 0.2 分钱一轮，且相同前缀会命中缓存更便宜。
# 自查工具：在命令行运行 Chat.exe --dumprequest，会导出“这一轮实际发出去的内容”。
max_history_messages = 20

# 截图快捷键（全局热键：在别的程序里按也能触发）
# 写法：修饰键 + 主键，修饰键必须至少有一个（Ctrl/Alt/Shift/Win），主键可用 A-Z、0-9、F1-F12。
# 例子：Ctrl+Alt+Q / Shift+F2 / Ctrl+Shift+A
# 如果这个组合被别的程序占用了，程序会自动改用内置默认 Ctrl+Alt+Q。
screenshot_hotkey = Ctrl+Alt+Q

# 显示 / 隐藏窗口的快捷键（全局热键）
# 窗口藏起来、托盘图标又被 Windows 折进“隐藏的图标”里时，按它最省事。
window_hotkey = Ctrl+Alt+C

# 是否在任务栏显示图标（默认 on）
#   on  = 任务栏和 Alt+Tab 里都有 Chat。
#   off = 任务栏和 Alt+Tab 里都不出现，改由右下角托盘图标操作：
#         双击托盘图标 = 显示 / 隐藏窗口
#         右键托盘图标 = 菜单（显示/隐藏、截图框选、查看完整对话、退出）
#   注意：关掉任务栏图标后，窗口最小化就没有任务栏按钮可以点回来，
#         所以程序会自动创建一个托盘图标兜底。托盘图标是自绘的蓝色对话气泡。
show_in_taskbar = off

# 点窗口右上角叉号（或按 Alt+F4）的行为：
#   hide = 只是隐藏到后台，程序继续运行（默认）。想叫回来：Ctrl+Alt+C，
#          或双击托盘图标；真正的退出在“右键窗口/托盘图标 → 退出 Chat”。
#   exit = 直接退出程序（老行为）
close_button = hide

# 窗口透明度（百分比，20~100）：
#   100 = 完全不透明（默认，字最清晰）
#    85 = 微微透，能隐约看见窗口后面的内容
#    70 = 明显透视，适合把窗口浮在资料上面边看边问
#    20 = 几乎只剩一层影子（下限，再低就等于把自己藏没了，程序会兜到 20）
# 说明：设为半透明后窗口就不再是不透明层，Windows 会关掉 ClearType，
#       字会比 100% 时略微发虚一点 —— 这是半透明窗口的固有代价，不是程序问题。
window_opacity = 100

# 加强提示词遵守（可选，默认 off）
# 如果发现模型偶尔不按 提示词.txt 的风格回答（例如又开始长篇解析），
# 把下面改成 on：程序会在提问前面再补一条 system 提醒，让规则更靠近问题。
# 代价是每轮多约 150 token，几乎可以忽略。
reinforce_prompt = off

# 系统提示词（决定模型的说话风格）
# 注意：如果同目录存在“提示词.txt”，以那个文件为准，这一行会被忽略。
#       写长提示词/背景说明请用 提示词.txt，这里只适合写一句话。
system_prompt = 你是一个乐于助人的中文助手。回答直给结论、条理清晰、不啰嗦。
